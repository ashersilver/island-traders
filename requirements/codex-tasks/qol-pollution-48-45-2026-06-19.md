### 2 — New module: `island_traders/engine/qol.py`

Create a standalone module (analogous to `flu.py`) with pure functions:

```python
from ..constants import (
    QOL_WEIGHT_FOOD, QOL_WEIGHT_HEALTH,
    QOL_WEIGHT_POLLUTION, QOL_WEIGHT_FOREST,
    OIL_POLLUTION_SCALE,
    POLLUTION_HEALTH_MITIGATION,
)

def food_coverage(demanded: int, bought: int) -> float:
    """Fraction of annual food demand met by households. Clamp [0, 1]."""
    if demanded <= 0:
        return 1.0
    return min(1.0, bought / demanded)

def health_coverage(demanded: int, bought: int) -> float:
    """Fraction of annual health-services demand met. Clamp [0, 1]."""
    if demanded <= 0:
        return 1.0
    return min(1.0, bought / demanded)

def raw_pollution_index(oil_consumed: int, population: int) -> float:
    """Un-mitigated pollution index in [0, 1].

    OIL_POLLUTION_SCALE is the oil consumption per person that would produce
    a pollution index of 1.0.
    """
    if population <= 0:
        return 0.0
    scale = OIL_POLLUTION_SCALE * max(1, population)
    return min(1.0, oil_consumed / scale)

def mitigated_pollution_index(
    oil_consumed: int,
    population: int,
    health_cov: float,
) -> float:
    """Pollution index after health-coverage mitigation.

    High health coverage (doctors/nurses) reduces the effective pollution impact
    by up to POLLUTION_HEALTH_MITIGATION (e.g. 50%).
    """
    raw = raw_pollution_index(oil_consumed, population)
    mitigation = health_cov * POLLUTION_HEALTH_MITIGATION
    return max(0.0, raw * (1.0 - mitigation))

def compute_qol(
    food_cov: float,
    health_cov: float,
    pollution_idx: float,
    forest_coverage: float = 0.0,  # stub until #159 lands
) -> float:
    """Composite QoL score in [0, 1]."""
    score = (
        food_cov        * QOL_WEIGHT_FOOD
        + health_cov    * QOL_WEIGHT_HEALTH
        + (1.0 - pollution_idx) * QOL_WEIGHT_POLLUTION
        + forest_coverage       * QOL_WEIGHT_FOREST
    )
    return max(0.0, min(1.0, score))
```

### 4 — Year-end QoL computation and population effects in `game.py`

Replace the current year-end population block (currently around line 389 —
`birth_rate = BASE_BIRTH_RATE * (1 - wealth_ratio)`) with the following logic.
Call a new method `_process_year_end_qol_and_population(year, prices, year_end_tick)`:

```python
def _process_year_end_qol_and_population(self, year: int, prices: dict, year_end_tick: int) -> None:
    from .qol import food_coverage, health_coverage, mitigated_pollution_index, compute_qol

    # Step 1: Compute QoL for every island.
    qol_scores: dict[int, float] = {}
    for player in self.players:
        fc   = food_coverage(
            getattr(player, "_food_demanded_this_year", 0),
            getattr(player, "_food_bought_this_year",   0),
        )
        hc   = health_coverage(
            getattr(player, "_health_demanded_this_year", 0),
            getattr(player, "_health_bought_this_year",   0),
        )
        pi   = mitigated_pollution_index(
            getattr(player, "_oil_consumed_this_year", 0),
            player.population,
            hc,
        )
        qol  = compute_qol(fc, hc, pi)
        # Store on player for game_state serialisation.
        player._qol_score         = round(qol, 3)
        player._food_coverage     = round(fc,  3)
        player._health_coverage   = round(hc,  3)
        player._pollution_index   = round(pi,  3)
        qol_scores[player.player_id] = qol
        self.io.print(
            f"  [QoL] {player.name}: food={fc:.0%} health={hc:.0%} "
            f"pollution={pi:.0%} → QoL {qol:.2f}"
        )

    # Step 2: QoL-modulated birth rate (replace wealth-only formula).
    for player in self.players:
        qol = qol_scores[player.player_id]
        birth_modifier = QOL_BIRTH_RATE_MIN + qol * (QOL_BIRTH_RATE_MAX - QOL_BIRTH_RATE_MIN)
        birth_rate = BASE_BIRTH_RATE * birth_modifier
        new_people = max(0, round(player.population * birth_rate))
        if new_people:
            player.population += new_people
            self.io.print(
                f"  [Population] {player.name}: +{new_people} people born "
                f"(QoL {qol:.2f} → rate {birth_rate:.3f}; "
                f"population now {player.population})"
            )

    # Step 3: Emigration — bottom-QoL islands lose residents to top-QoL.
    if len(self.players) < 2:
        return
    mean_qol = sum(qol_scores.values()) / len(qol_scores)
    best_player = max(self.players, key=lambda p: qol_scores[p.player_id])
    for player in self.players:
        if player is best_player:
            continue
        qol = qol_scores[player.player_id]
        if mean_qol - qol >= QOL_EMIGRATION_THRESHOLD:
            emigrants = max(1, round(player.population * QOL_EMIGRATION_RATE))
            emigrants = min(emigrants, player.population - 1)  # never empty an island
            if emigrants > 0:
                player.population     -= emigrants
                best_player.population += emigrants
                self.io.print(
                    f"  [Emigration] {emigrants} people left {player.name} "
                    f"(QoL {qol:.2f}) for {best_player.name} "
                    f"(QoL {qol_scores[best_player.player_id]:.2f})."
                )
```

Remove the old `for player, wealth in zip(...)` birth-rate block and call
